```dataview
TABLE WITHOUT ID
  file.link AS "Fecha / Apunte",
  default(etapa, "General") AS "Etapa o Tema",
  file.mtime AS "Última edición"
FROM "Diario"
WHERE file.name != "00_Diario" AND !contains(file.name, "Revision") AND !contains(file.name, "Fuentes") AND !contains(file.name, "Machete")
SORT file.name DESC
LIMIT 5
```
